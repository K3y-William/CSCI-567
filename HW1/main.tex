\section*{Problem 1: linear algebra and gradients warm-up}

\subsection*{1.1}
% gradient of g(w) = b^T w + 1/2 w^T A w

\[
\nabla_{\vect{w}}\, g(\vect{w})
= \nabla_{\vect{w}}\big(\vect{b}^\top \vect{w}\big) + \tfrac12\, \nabla_{\vect{w}}\big(\vect{w}^\top A \vect{w}\big)
= \vect{b} + \tfrac12\big(A\vect{w} + A^\top \vect{w}\big)
\]
Since $A$ is symmetric, $A^\top = A$, so
\[
\nabla_{\vect{w}}\, g(\vect{w}) = \vect{b} + A\vect{w}.
\]

\vspace{1cm}

\subsection*{1.2}
% rank(x x^T)

\[
\vect{x}\vect{x}^\top =
\begin{bmatrix} x_1 \\ x_2 \\ \vdots \\ x_n \end{bmatrix}
\begin{bmatrix} x_1 & x_2 & \cdots & x_n \end{bmatrix}
=
\begin{bmatrix}
x_1 x_1 & x_1 x_2 & \cdots & x_1 x_n \\
x_2 x_1 & x_2 x_2 & \cdots & x_2 x_n \\
\vdots  & \vdots  & \ddots & \vdots  \\
x_n x_1 & x_n x_2 & \cdots & x_n x_n
\end{bmatrix}
=
\begin{bmatrix} x_1 \vect{x}^\top \\ x_2 \vect{x}^\top \\ \vdots \\ x_n \vect{x}^\top \end{bmatrix}.
\]
Every row is a scalar multiple of $\vect{x}^\top$, so there is only one linearly independent row.
Since $\vect{x} \neq \vect{0}$, $\operatorname{rank}(\vect{x}\vect{x}^\top) = 1$.

\subsection*{1.3}
% tr(AB) = tr(BA)

$AB \in \R^{m\times m}$ with $(AB)_{ii} = \sum_{j=1}^{n} A_{ij}B_{ji}$, and
$BA \in \R^{n\times n}$ with $(BA)_{jj} = \sum_{i=1}^{m} B_{ji}A_{ij}$. Hence
\[
\operatorname{tr}(AB)
= \sum_{i=1}^{m} (AB)_{ii}
= \sum_{i=1}^{m} \sum_{j=1}^{n} A_{ij} B_{ji}
= \sum_{j=1}^{n} \sum_{i=1}^{m} B_{ji} A_{ij}
= \sum_{j=1}^{n} (BA)_{jj}
= \operatorname{tr}(BA),
\]
\vspace{1cm}

\subsection*{1.4}
% A = [[2,1],[1,2]] is PSD

Let $\vect{x} = \begin{bmatrix} x_1 \\ x_2 \end{bmatrix}$. Then
\begin{align*}
\vect{x}^\top A \vect{x}
&= \begin{bmatrix} x_1 & x_2 \end{bmatrix}
   \begin{bmatrix} 2 & 1 \\ 1 & 2 \end{bmatrix}
   \begin{bmatrix} x_1 \\ x_2 \end{bmatrix}
 = \begin{bmatrix} 2x_1 + x_2 & x_1 + 2x_2 \end{bmatrix}
   \begin{bmatrix} x_1 \\ x_2 \end{bmatrix} \\
&= (2x_1 + x_2)x_1 + (x_1 + 2x_2)x_2 \\
&= 2x_1^2 + 2x_1x_2 + 2x_2^2 \\
&= (x_1 + x_2)^2 + x_1^2 + x_2^2 .
\end{align*}
Each of the three terms is a square, hence $\geq 0$, so $\vect{x}^\top A \vect{x} \geq 0$ for all
$\vect{x} \in \R^2$. Therefore $A$ is positive semidefinite.

\section*{Problem 3: how much can you trust a test-set number?}

\subsection*{3.1 Sizing a test set}
Hoeffding's inequality gives
\[
P\big(|\bar{Z} - \mu| \geq \epsilon\big) \;\leq\; 2e^{-2n\epsilon^2}.
\]
We want this failure probability to be at most $\delta$:
\[
2e^{-2n\epsilon^2} \leq \delta
\;\iff\; -2n\epsilon^2 \leq \ln\frac{\delta}{2}
\;\iff\; n \;\geq\; \frac{\ln(2/\delta)}{2\epsilon^2}.
\]
With $\delta = 0.05$, $\ln(2/\delta) \approx 3.689$, so
\[
\epsilon = 0.01:\quad n \geq \frac{3.689}{2(0.01)^2} \approx 18{,}445,
\qquad
\epsilon = 0.001:\quad n \geq \frac{3.689}{2(0.001)^2} \approx 1{,}844{,}440
\]
Adding $\epsilon$ one more decimal digit of certified precision required test set size multiply by $100$.

\vspace{1cm}

\subsection*{3.2 Two test sets, two intervals}

\paragraph{(i)}
\begin{align*}
2e^{-2n_j\epsilon_j^2} &= \delta_j \\
e^{-2n_j\epsilon_j^2} &= \frac{\delta_j}{2}  = 0.025 \\
-2n_j\epsilon_j^2 &= \ln(0.025) \\
\epsilon_j^2 &= \frac{\ln(1/0.025)}{2n_j} = \frac{\ln 40}{2n_j}  \\
\epsilon_j &= \sqrt{\frac{\ln 40}{2n_j}} .
\end{align*}

\[
\epsilon_1 = \sqrt{\frac{\ln 40}{2\cdot 50{,}000}} \approx 0.0061 \;,
\qquad
\epsilon_2 = \sqrt{\frac{\ln 40}{2\cdot 10{,}000}} \approx 0.0136 \;.
\]

\vspace{1cm}

\paragraph{(ii)}

\textbf{(a)} Let $E_j$ be the event that interval $j$ does not contain $\mu$, \ $|A_j - \mu| > \epsilon_j$.
By Hoeffding, $P(E_j) \leq \delta_j$. Simultaneous validity fails when $E_1 \cup E_2$ occurs,
so by the union bound
\[
P(\text{not simultaneously valid}) = P(E_1 \cup E_2) \;\leq\; P(E_1) + P(E_2) \;\leq\; \delta_1 + \delta_2 .
\]

\textbf{(b)} Suppose both intervals are valid, i.e.\ $|A_1 - \mu| \leq \epsilon_1$ and $|A_2 - \mu| \leq \epsilon_2$.
So $\mu$ lies in both intervals:
\[
|A_1 - A_2| = |(A_1 - \mu) + (\mu - A_2)| \;\leq\; |A_1 - \mu| + |\mu - A_2| \;\leq\; \epsilon_1 + \epsilon_2 .
\]

\section*{Problem 4: why mini-batches are less noisy than single-example SGD}

\subsection*{4.1 One example}

An expectation over $I$ is an average over $i$.

\textbf{Unbiasedness}
\[
\E[\vect{g}_I] = \sum_{i=1}^{n} \frac{1}{n}\, \vect{g}_i = \vect{g}.
\]

\textbf{Noise}
\[
\E\big[\|\vect{g}_I - \vect{g}\|^2\big] = \frac{1}{n}\sum_{i=1}^{n} \|\vect{g}_i - \vect{g}\|^2 = \sigma^2 .
\]

\vspace{1cm}

\subsection*{4.2 A mini-batch}

Each $I_b$ is uniform on $\{1,\dots,n\}$, so by 4.1, $\E[\vect{g}_{I_b}] = \vect{g}$ and
$\E\big[\|\vect{g}_{I_b} - \vect{g}\|^2\big] = \sigma^2$ for every $b$.

\textbf{Unbiasedness} 
\[
\E[\hat{\vect{g}}_B] = \frac{1}{B}\sum_{b=1}^{B} \E[\vect{g}_{I_b}] = \frac{1}{B}\sum_{b=1}^{B} \vect{g} = \vect{g}.
\]

\textbf{Noise}
\[
\hat{\vect{g}}_B - \vect{g} = \frac{1}{B}\sum_{b=1}^{B} \big(\vect{g}_{I_b} - \vect{g}\big).
\]
Expanding the squared norm,
\[
\|\hat{\vect{g}}_B - \vect{g}\|^2
= \frac{1}{B^2}\Big\| \sum_{b=1}^{B} \big(\vect{g}_{I_b} - \vect{g}\big) \Big\|^2
= \frac{1}{B^2}\Big( \sum_{b=1}^{B} \|\vect{g}_{I_b} - \vect{g}\|^2
  + \sum_{b \neq c} \big\langle \vect{g}_{I_b} - \vect{g},\; \vect{g}_{I_c} - \vect{g} \big\rangle \Big).
\]
For $b \neq c$, $I_b$ and $I_c$ are independent, so $\vect{g}_{I_b} - \vect{g}$ and $\vect{g}_{I_c} - \vect{g}$
are independent and
\[
\E\big[\langle \vect{g}_{I_b} - \vect{g},\; \vect{g}_{I_c} - \vect{g} \rangle\big]
= \big\langle \E[\vect{g}_{I_b}] - \vect{g},\; \E[\vect{g}_{I_c}] - \vect{g} \big\rangle
= \langle \vect{g} - \vect{g},\; \vect{g} - \vect{g} \rangle
= 0 .
\]
Therefore:
\[
\E\big[\|\hat{\vect{g}}_B - \vect{g}\|^2\big]
= \frac{1}{B^2}\sum_{b=1}^{B} \E\big[\|\vect{g}_{I_b} - \vect{g}\|^2\big]
= \frac{1}{B^2}\cdot B\sigma^2
= \frac{\sigma^2}{B}.
\]

\begin{itemize}
    \item Gradient evaluations per step: GD uses $n = 1000$ example gradients, SGD uses 1.
    \item Total passes over the data: GD uses $20$ passes, SGD uses $1$ pass.
    \item Curve shape: GD is smooth because each step uses the exact gradient. SGD is jagged because each step uses one random example's gradient, which is unbiased but has noise, so individual steps can increase $\emprisk$.
    \item GD ($\eta = 0.5$) reached $0.117533$, essentially the optimum, however for the best SGD

\end{itemize}
